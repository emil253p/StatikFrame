# StatikQuest on Steam Frame: settings. The launcher reads the copy of this file in
# ~/.local/share/statikquest/settings.txt (it makes it the first time). Quit and relaunch the
# game after changing it.

# Where the extracted game is (the folder with eboot.bin, sce_sys and sce_module in it).
game=~/Games/Statik/CUSA06929

# Frame rate: 60 (the game's own, as on PlayStation VR), 90 or 120. Higher needs much more of
# the headset's chip: start with 60.
frame_rate=60
# Per-eye sharpness, a percentage of the 960x1080 PSVR panel: game (145% base PS4, 180% Pro),
# 180, 220 or 260. On the headset's own chip, start with the game's own.
resolution=game
# PS4 Pro mode (1) draws sharper and costs more; 0 is a base PS4.
pro=0

# Field of view: 70..120 percent of psvr's (or of the headset's, with fov_of=headset).
fov=110
fov_of=psvr

# Both motion controllers hold the game's controller together (1), or only one (0).
two_hands=1
# The hand that holds it alone: right or left.
controller_hand=right
# The right stick: both (the PS4's right stick and a finger on the touchpad), stick, or touchpad.
right_stick=both

# The console's language: system (the headset's), or a tag such as en-US, de-DE, fr-FR.
language=system
# Sharpening of the picture in the headset, 0 to 1.
sharpen=0.3
# Seconds to wait for the headset before giving up.
wait=30
# GPU work in smaller pieces (draws per piece; 0 = a whole frame at once), and MSAA at most.
flush_draws=64
msaa=1
